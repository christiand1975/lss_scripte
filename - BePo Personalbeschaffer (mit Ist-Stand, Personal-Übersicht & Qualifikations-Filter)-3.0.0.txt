/* global $, loadedBuildings */

(function () {
  'use strict';

  const minimalRemainingPersonnelInPoliceStation = 34;

  // Pseudo-Key für Personal ohne jede Ausbildung (weder fertig noch im Unterricht) – BePo-Ist-Stand
  const NO_EDUCATION_KEY = '__no_education__';

  // ── BePo-spezifisch: Soll-Werte pro Qualifikation (nur für Auto-Anwerbung auf BePo-Wachen) ──
  // key: null = Personal ohne jede Ausbildung
  const personnelSettingsInternal = [
    { caption: "Ohne Ausbildung",            key: null,                     numberOfRequiredPersonnel: 83  },
    { caption: "Hundeführer (Schutzhund)",   key: "k9",                     numberOfRequiredPersonnel: 6   },
    { caption: "Hundertschaftsführer (FüKw)",key: "police_fukw",            numberOfRequiredPersonnel: 15  },
    { caption: "MEK",                        key: "police_mek",             numberOfRequiredPersonnel: 42  },
    { caption: "SEK",                        key: "police_sek",             numberOfRequiredPersonnel: 42  },
    { caption: "Reiterstaffel",              key: "police_horse",           numberOfRequiredPersonnel: 24  },
    { caption: "Wasserwerfer",               key: "police_wasserwerfer",    numberOfRequiredPersonnel: 15  },
    { caption: "Zugführer (leBefKw)",        key: "police_einsatzleiter",   numberOfRequiredPersonnel: 12  },
    { caption: "Lautsprecheroperator",       key: "police_speaker_operator",numberOfRequiredPersonnel: 5   },
  ];

  const originalRequiredByKey = {};
  personnelSettingsInternal.forEach(s => { originalRequiredByKey[s.key] = s.numberOfRequiredPersonnel; });

  // Effektiver Bedarf nach Abzug des Ist-Stands (BePo)
  const effectiveNeeds = {};

  const personnelSettingsProxy = personnelSettingsInternal.map(setting => {
    return new Proxy({ ...setting, numberOfSelectedPersonnel: 0 }, {
      set(target, prop, value) {
        target[prop] = value;
        if (prop === 'numberOfSelectedPersonnel') {
          updateFooter(target.key, value);
        }
        return true;
      }
    });
  });

  // ── Bekannte, schöne Bezeichnungen für Qualifikations-Keys (Fallback: automatisch generiert) ──
  const KNOWN_CAPTIONS = {
    k9: "Hundeführer (Schutzhund)",
    police_fukw: "Hundertschaftsführer (FüKw)",
    police_mek: "MEK",
    police_sek: "SEK",
    police_horse: "Reiterstaffel",
    police_wasserwerfer: "Wasserwerfer",
    police_einsatzleiter: "Zugführer (leBefKw)",
    police_speaker_operator: "Lautsprecheroperator",
  };

  // ─── HTML-Fragment-Cache ───────────────────────────────────────────────────
  const hrefHtmlCache = new Map(); // href -> Promise<string html>

  function fetchHrefCached(href) {
    if (!hrefHtmlCache.has(href)) {
      hrefHtmlCache.set(href, $.get(href));
    }
    return hrefHtmlCache.get(href);
  }

  // ─── Ist-Stand der BePo-Wache laden (nur BePo-Auto-Anwerbung) ─────────────

  async function fetchIstStand(buildingId) {
    const html = await $.get(`/buildings/${buildingId}/personals`);
    const doc = new DOMParser().parseFromString(html, 'text/html');
    const rows = doc.querySelectorAll('#personal_table tr[data-filterable-by]');
    const counts = {};

    rows.forEach(row => {
      let keys = [];
      try {
        keys = JSON.parse(row.dataset.filterableBy);
      } catch (e) { /* leer */ }
      keys.forEach(k => { counts[k] = (counts[k] || 0) + 1; });

      const eduSpan = row.querySelector('[data-education-key]');
      if (eduSpan) {
        const k = eduSpan.dataset.educationKey;
        counts[k] = (counts[k] || 0) + 1;
      }

      if (keys.length === 0 && !eduSpan) {
        counts[NO_EDUCATION_KEY] = (counts[NO_EDUCATION_KEY] || 0) + 1;
      }
    });

    return counts;
  }

  async function initEffectiveNeeds(buildingId) {
    let istStand = {};
    try {
      istStand = await fetchIstStand(buildingId);
      console.log('[BePo DEBUG] Ist-Stand geladen:', istStand);
    } catch (e) {
      console.warn('[BePo] Fehler beim Laden des Ist-Stands:', e);
    }

    personnelSettingsInternal.forEach(setting => {
      const idKey = setting.key === null ? 'null' : setting.key;
      const ist = setting.key ? (istStand[setting.key] || 0) : (istStand[NO_EDUCATION_KEY] || 0);
      const noch = Math.max(0, setting.numberOfRequiredPersonnel - ist);
      effectiveNeeds[setting.key] = { required: noch, selected: 0 };
      console.log(`[BePo DEBUG] ${setting.caption}: Soll=${setting.numberOfRequiredPersonnel}, Ist=${ist}, Noch=${noch}`);

      const istEl = document.getElementById(`ist-stand-${idKey}`);
      if (istEl) istEl.innerHTML = ist;

      const proxySetting = personnelSettingsProxy.find(s => s.key === setting.key);
      if (proxySetting) proxySetting.numberOfRequiredPersonnel = noch;

      const reqEl = document.getElementById(`number-of-required-personnel-${idKey}`);
      if (reqEl) reqEl.innerHTML = noch;
    });

    console.log('[BePo DEBUG] initEffectiveNeeds abgeschlossen');
  }

  // ─── DOM-Initialisierung (generisch, für alle Wachenarten) ────────────────

  function initPanelBodies() {
    Array.from(document.getElementsByClassName('panel-body'))
      .forEach(el => el.classList.add('hidden'));
  }

  function removePanelHeadingClickEvent() {
    Array.from(document.getElementsByClassName('personal-select-heading'))
      .forEach(el => {
        const clone = el.cloneNode(true);
        el.replaceWith(clone);
        clone.addEventListener('click', panelHeadingClickEvent);
      });
  }

  /**
   * Fügt pro gelisteter Wache eine Button-Gruppe hinzu.
   * includeHireControls=true (nur BePo): zusätzlich Reset- & Anwerben-Button.
   * Der Info-Button (Personal-Übersicht) wird immer hinzugefügt.
   */
  function addPersonnelSelectors(includeHireControls) {
    Array.from(document.getElementsByClassName('panel-heading personal-select-heading'))
      .forEach(el => {
        const buildingId = el.getAttribute('building_id');
        const group = document.createElement('div');
        group.classList.add('btn-group', 'btn-group-xs');
        group.setAttribute('role', 'group');

        if (includeHireControls) {
          group.appendChild(createResetButton(buildingId));
          group.appendChild(createSelectButton(buildingId));
        }
        group.appendChild(createInfoButton(buildingId));

        el.children[1].prepend(group);
      });
  }

  // ─── Footer (nur BePo) ─────────────────────────────────────────────────────

  function addFooter() {
    const wrapper = document.createElement('div');
    wrapper.style = 'display:flex;flex-wrap:wrap;flex-direction:row;column-gap:15px';

    const list = document.createElement('ul');
    list.classList.add('list-inline');
    list.style = 'color:#fff;padding-top:8px;';

    personnelSettingsProxy.forEach(s => list.appendChild(createTotalSummaryElement(s)));
    wrapper.appendChild(list);

    const nav = document.querySelector('.navbar.navbar-default.navbar-fixed-bottom');
    nav.children[0].children[0].insertAdjacentElement('afterend', wrapper);
  }

  function updateFooter(key, selectedPersonnel) {
    const idKey = key === null ? 'null' : key;
    const elSelected = document.getElementById(`number-of-selected-personnel-${idKey}`);
    if (elSelected) elSelected.innerHTML = selectedPersonnel;

    const need = effectiveNeeds[key];
    const required = need ? need.required : (personnelSettingsProxy.find(s => s.key === key) || {}).numberOfRequiredPersonnel;
    const labelClass = selectedPersonnel === required ? 'label-success' : 'label-warning';
    const spanPersonnel = document.getElementById(`personnel-${idKey}`);
    if (spanPersonnel) {
      spanPersonnel.classList.remove('label-success', 'label-warning');
      spanPersonnel.classList.add(labelClass);
    }
  }

  function createTotalSummaryElement(setting) {
    const idKey = setting.key === null ? 'null' : setting.key;
    const li = document.createElement('li');

    const spanCaption = document.createElement('span');
    spanCaption.innerHTML = `${setting.caption} (Soll ${originalRequiredByKey[setting.key]}): `;

    const spanIst = document.createElement('span');
    spanIst.id = `ist-stand-${idKey}`;
    spanIst.innerHTML = '…';
    spanIst.style = 'color:#9fd6ff;font-weight:bold;';

    const spanIstLabel = document.createElement('span');
    spanIstLabel.innerHTML = ' vorhanden, ';

    const spanSelected = document.createElement('span');
    spanSelected.id = `number-of-selected-personnel-${idKey}`;
    spanSelected.innerHTML = '0';

    const spanRequired = document.createElement('span');
    spanRequired.id = `number-of-required-personnel-${idKey}`;
    spanRequired.innerHTML = '…';

    const spanPersonnel = document.createElement('span');
    spanPersonnel.id = `personnel-${idKey}`;
    spanPersonnel.classList.add('label', 'label-warning');
    spanPersonnel.appendChild(spanSelected);
    spanPersonnel.appendChild(document.createTextNode('/'));
    spanPersonnel.appendChild(spanRequired);

    const spanAngeworbenLabel = document.createElement('span');
    spanAngeworbenLabel.innerHTML = ' angeworben (Rest)';

    li.appendChild(spanCaption);
    li.appendChild(spanIst);
    li.appendChild(spanIstLabel);
    li.appendChild(spanPersonnel);
    li.appendChild(spanAngeworbenLabel);
    return li;
  }

  // ─── Generische Personal-Übersicht & Qualifikations-Filter (alle Wachenarten) ─

  // key -> Anzeigename. Wird beim Auszählen automatisch befüllt (unbekannte
  // Keys werden hübsch formatiert, bekannte Keys nutzen KNOWN_CAPTIONS).
  const discoveredQualifications = new Map();

  // buildingId -> { counts: { key: anzahlFrei }, noQual: anzahlOhneQualifikation }
  const buildingPersonnelCounts = new Map();

  function prettifyKey(key) {
    return key
      .split('_')
      .filter(Boolean)
      .map(w => w.charAt(0).toUpperCase() + w.slice(1))
      .join(' ');
  }

  function getCaptionForKey(key) {
    if (KNOWN_CAPTIONS[key]) return KNOWN_CAPTIONS[key];
    return prettifyKey(key);
  }

  function registerQualification(key) {
    if (!discoveredQualifications.has(key)) {
      discoveredQualifications.set(key, getCaptionForKey(key));
    }
  }

  /**
   * Zählt in einem geladenen HTML-Fragment (Personal-Auswahlliste einer Wache,
   * beliebige Wachenart), wie viele FREIE (nicht an ein Fahrzeug gebundene)
   * Personen welche Qualifikation haben. Qualifikationen werden automatisch
   * anhand der Checkbox-Attribute mit Wert "true" erkannt.
   *
   * Annahmen (wie im übrigen Skript schon genutzt):
   *  - Checkbox pro Person: input.schooling_checkbox mit Attributen wie
   *    police_mek="true" usw.
   *  - "An Fzg gebunden"-Info steht in der 4. Zelle der Zeile (children[3])
   *  - Schulungs-Zelle hat id="school_personal_education_..."
   *
   * Gibt zurück: { counts: {key: N}, noQual: N }
   */
  function countPersonnelGeneric(html) {
    const doc = new DOMParser().parseFromString(html, 'text/html');
    const counts = {};
    let noQual = 0;

    const checkboxes = doc.querySelectorAll('input.schooling_checkbox');
    checkboxes.forEach(cb => {
      const tr = cb.closest('tr');
      if (!tr) return;

      const assignedCell = tr.children[3];
      const isBoundToVehicle = !!assignedCell && assignedCell.innerHTML.replace(/\s/g, '').length > 0;
      if (isBoundToVehicle) return; // an Fzg gebundenes Personal komplett ausblenden

      let matchedAny = false;
      Array.from(cb.attributes).forEach(attr => {
        if (attr.value === 'true') {
          registerQualification(attr.name);
          counts[attr.name] = (counts[attr.name] || 0) + 1;
          matchedAny = true;
        }
      });

      const schoolCell = tr.querySelector("td[id^='school_personal_education_']");
      const inSchool = !!schoolCell && schoolCell.innerHTML.replace(/\s/g, '').length > 0;

      if (!matchedAny && !inSchool) {
        noQual++;
      }
    });

    return { counts, noQual };
  }

  function buildOverviewElement(buildingId, data) {
    const div = document.createElement('div');
    div.id = `personnel-overview-${buildingId}`;
    div.classList.add('personnel-overview');
    div.style = 'padding:6px 15px;background:#f5f5f5;border-top:1px solid #ddd;font-size:12px;color:#333;';

    const entries = Object.keys(data.counts)
      .map(key => ({ caption: getCaptionForKey(key), count: data.counts[key] }))
      .sort((a, b) => a.caption.localeCompare(b.caption, 'de'));

    if (data.noQual > 0) {
      entries.push({ caption: 'Ohne Qualifikation', count: data.noQual });
    }

    if (entries.length === 0) {
      div.innerHTML = '<em>Kein freies Personal gefunden.</em>';
    } else {
      div.innerHTML = entries
        .map(e => `<span style="margin-right:14px;display:inline-block;"><strong>${e.caption}:</strong> ${e.count}</span>`)
        .join('');
    }

    return div;
  }

  function getHrefForBuilding(buildingId) {
    const heading = getPanelHeading(buildingId);
    const match = heading.outerHTML.match(/href="([^"]+)"/);
    return match ? match[1] : null;
  }

  /**
   * Lädt (bzw. nutzt aus dem Cache) das Personal-Fragment einer Wache,
   * zählt die Qualifikationen aus, speichert das Ergebnis und blendet die
   * Übersicht direkt unter deren Überschrift ein. Erneuter Aufruf blendet
   * eine bereits vorhandene Übersicht nur ein/aus, ohne neu zu laden.
   */
  async function showPersonnelOverview(buildingId, triggerBtn) {
    const existing = document.getElementById(`personnel-overview-${buildingId}`);
    if (existing) {
      existing.classList.toggle('hidden');
      return;
    }

    if (triggerBtn) triggerBtn.disabled = true;

    const href = getHrefForBuilding(buildingId);
    if (!href) {
      console.warn('[BePo] Konnte href für Wache', buildingId, 'nicht ermitteln');
      if (triggerBtn) triggerBtn.disabled = false;
      return;
    }

    let html;
    try {
      html = await fetchHrefCached(href);
    } catch (e) {
      console.warn('[BePo] Fehler beim Laden der Personal-Übersicht für Wache', buildingId, e);
      if (triggerBtn) triggerBtn.disabled = false;
      return;
    }

    const data = countPersonnelGeneric(html);
    buildingPersonnelCounts.set(buildingId, data);

    const overviewEl = buildOverviewElement(buildingId, data);
    const heading = getPanelHeading(buildingId);
    heading.parentElement.insertBefore(overviewEl, heading.nextSibling);

    refreshQualificationDropdown();

    if (triggerBtn) triggerBtn.disabled = false;
  }

  function createInfoButton(buildingId) {
    const icon = document.createElement('span');
    icon.classList.add('glyphicon', 'glyphicon-info-sign');

    const btn = document.createElement('button');
    btn.classList.add('btn', 'btn-xs', 'btn-default', 'personnel-info-button');
    btn.id = `personnel-info-button-${buildingId}`;
    btn.setAttribute('type', 'button');
    btn.setAttribute('data-building-id', buildingId);
    btn.title = 'Personal-Übersicht dieser Wache anzeigen';
    btn.appendChild(icon);
    btn.addEventListener('click', infoButtonClick);
    return btn;
  }

  async function infoButtonClick(event) {
    event.preventDefault();
    event.stopPropagation();
    const btn = event.target.closest('button');
    const buildingId = btn.dataset.buildingId;
    await showPersonnelOverview(buildingId, btn);
  }

  // Referenz auf das Dropdown-Element, damit wir es später befüllen können
  let qualificationSelectEl = null;

  function addQualificationToolbar() {
    const wrapper = document.createElement('div');
    wrapper.style = 'display:flex;align-items:center;flex-wrap:wrap;gap:10px;margin:10px 15px;';

    const loadAllBtn = document.createElement('button');
    loadAllBtn.type = 'button';
    loadAllBtn.classList.add('btn', 'btn-xs', 'btn-info');
    loadAllBtn.innerText = 'Personal-Übersicht aller Wachen laden';

    const select = document.createElement('select');
    select.classList.add('form-control', 'input-sm');
    select.style = 'width:auto;display:inline-block;';
    select.disabled = true;

    const placeholderOption = document.createElement('option');
    placeholderOption.value = '';
    placeholderOption.innerText = 'Erst "Alle laden" klicken …';
    select.appendChild(placeholderOption);

    select.addEventListener('change', () => {
      filterPanelsByQualification(select.value || null);
    });

    qualificationSelectEl = select;

    loadAllBtn.addEventListener('click', async () => {
      loadAllBtn.disabled = true;
      const originalText = loadAllBtn.innerText;
      const headings = Array.from(document.getElementsByClassName('panel-heading personal-select-heading'));

      for (let i = 0; i < headings.length; i++) {
        const buildingId = headings[i].getAttribute('building_id');
        loadAllBtn.innerText = `Lade Wache ${i + 1}/${headings.length} …`;
        await showPersonnelOverview(buildingId, null);
        await new Promise(r => setTimeout(r, 150)); // kleine Pause, um den Server zu schonen
      }

      loadAllBtn.innerText = originalText;
      loadAllBtn.disabled = false;

      select.disabled = false;
      placeholderOption.innerText = 'Alle anzeigen';
      refreshQualificationDropdown();
    });

    wrapper.appendChild(loadAllBtn);
    wrapper.appendChild(select);

    const firstHeading = document.querySelector('.panel-heading.personal-select-heading');
    const panelGroup = firstHeading ? firstHeading.closest('.panel-group') : null;

    if (panelGroup && panelGroup.parentElement) {
      panelGroup.parentElement.insertBefore(wrapper, panelGroup);
    } else {
      document.body.prepend(wrapper);
    }
  }

  /**
   * Baut die Optionsliste des Dropdowns neu auf Basis aller bisher
   * entdeckten Qualifikationen (discoveredQualifications). Die aktuelle
   * Auswahl bleibt erhalten, falls sie noch existiert.
   */
  function refreshQualificationDropdown() {
    if (!qualificationSelectEl) return;

    const previousValue = qualificationSelectEl.value;
    const placeholderText = qualificationSelectEl.disabled
      ? 'Erst "Alle laden" klicken …'
      : 'Alle anzeigen';

    qualificationSelectEl.innerHTML = '';

    const placeholderOption = document.createElement('option');
    placeholderOption.value = '';
    placeholderOption.innerText = placeholderText;
    qualificationSelectEl.appendChild(placeholderOption);

    const sortedKeys = Array.from(discoveredQualifications.keys())
      .sort((a, b) => discoveredQualifications.get(a).localeCompare(discoveredQualifications.get(b), 'de'));

    sortedKeys.forEach(key => {
      const option = document.createElement('option');
      option.value = key;
      option.innerText = discoveredQualifications.get(key);
      qualificationSelectEl.appendChild(option);
    });

    if (sortedKeys.includes(previousValue)) {
      qualificationSelectEl.value = previousValue;
    }
  }

  /**
   * Blendet Wachen-Panels ein/aus, je nachdem ob sie freies Personal mit der
   * gewählten Qualifikation haben. key=null/'' → alle Wachen anzeigen.
   * Wachen, deren Personal noch nicht geladen wurde, werden bei aktivem
   * Filter sicherheitshalber ausgeblendet (kein Datenstand = nicht bestätigt).
   */
  function filterPanelsByQualification(key) {
    Array.from(document.getElementsByClassName('panel-heading personal-select-heading'))
      .forEach(heading => {
        const buildingId = heading.getAttribute('building_id');
        const panel = heading.closest('.panel') || heading.parentElement;
        if (!panel) return;

        if (!key) {
          panel.style.display = '';
          return;
        }

        const data = buildingPersonnelCounts.get(buildingId);
        const count = data && data.counts[key] ? data.counts[key] : 0;
        panel.style.display = count > 0 ? '' : 'none';
      });
  }

  // ─── BePo-spezifische Buttons (Auto-Anwerbung) ────────────────────────────

  function createResetButton(buildingId) {
    const trashIcon = document.createElement('span');
    trashIcon.classList.add('glyphicon', 'glyphicon-trash');

    const resetButton = document.createElement('button');
    resetButton.classList.add('btn', 'btn-xs', 'btn-default', 'personnel-reset-button');
    resetButton.setAttribute('type', 'button');
    resetButton.setAttribute('data-building-id', buildingId);
    resetButton.addEventListener('click', resetPersonnelClick);
    resetButton.appendChild(trashIcon);
    return resetButton;
  }

  function createSelectButton(buildingId) {
    const userIcon = document.createElement('span');
    userIcon.classList.add('glyphicon', 'glyphicon-user');

    const btn = document.createElement('button');
    btn.classList.add('btn', 'btn-xs', 'btn-default', 'personnel-select-button');
    btn.id = `personnel-select-button-${buildingId}`;
    btn.setAttribute('type', 'button');
    btn.setAttribute('data-building-id', buildingId);
    btn.addEventListener('click', selectPersonnelClick);
    btn.innerText = ' Personal auswählen';
    btn.prepend(userIcon);
    return btn;
  }

  // ─── Click-Handler (BePo Auto-Anwerbung) ──────────────────────────────────

  async function selectPersonnelClick(event) {
    event.preventDefault();
    const button = event.target.closest('button');
    button.disabled = true;
    button.classList.replace('btn-default', 'btn-success');
    const okIcon = document.createElement('span');
    okIcon.classList.add('glyphicon', 'glyphicon-ok');
    button.replaceChild(okIcon, button.children[0]);

    const buildingId = button.dataset.buildingId;
    await selectPersonnel(buildingId);

    const panelBody = getPanelBody(buildingId);
    const count = panelBody.querySelectorAll('input:checked').length;
    button.innerHTML += ` (${count})`;
  }

  async function resetPersonnelClick(event) {
    event.preventDefault();
    const resetButton = event.target.closest('button');
    const buildingId = resetButton.dataset.buildingId;
    document.getElementById(`personnel-select-button-${buildingId}`)
      .replaceWith(createSelectButton(buildingId));
    resetPersonnel(buildingId);
  }

  // ─── Kernlogik: Personal auswählen (BePo Auto-Anwerbung) ──────────────────

  async function selectPersonnel(buildingId) {
    await panelHeadingClick(buildingId);

    for (let i = personnelSettingsProxy.length - 1; i >= 0; i--) {
      const setting = personnelSettingsProxy[i];
      const need    = effectiveNeeds[setting.key];

      const required = need ? need.required : setting.numberOfRequiredPersonnel;

      if (required <= 0) continue;

      const panelBody = getPanelBody(buildingId);
      let inputElements = [];

      if (setting.key === null) {
        const cells = panelBody.querySelectorAll("td[id^='school_personal_education_']");
        cells.forEach(cell => {
          if (cell.innerHTML.replace(/\s/g, '').length === 0) {
            inputElements.push(cell.parentElement.children[0].children[0]);
          }
        });
      } else {
        inputElements = Array.from(panelBody.querySelectorAll(`input[${setting.key}='true']`));
      }

      inputElements = inputElements.filter(el => {
        if (!el) return false;
        return el.parentElement.parentElement.children[3].innerHTML.replace(/\s/g, '').length === 0;
      });

      let j = inputElements.length - 1;
      while (setting.numberOfSelectedPersonnel < required && j >= 0) {
        inputElements[j].click();
        j--;

        if (
          setting.key === null &&
          inputElements.length - setting.numberOfSelectedPersonnel <= minimalRemainingPersonnelInPoliceStation
        ) {
          break;
        }
      }
    }
  }

  function resetPersonnel(buildingId) {
    getPanelBody(buildingId).querySelectorAll('input:checked')
      .forEach(el => el.click());
  }

  // ─── Panel-Logik (generisch) ───────────────────────────────────────────────

  function addClickEventHandlerToCheckboxes() {
    Array.from(document.getElementsByClassName('schooling_checkbox'))
      .forEach(el => el.addEventListener('change', updateNumberOfSelectedPersonnel));
  }

  function updateNumberOfSelectedPersonnel(event) {
    const attrs = Object.keys(event.target.attributes);
    const attrIdx = attrs.find(k => event.target.attributes[k].value === 'true');
    const key = attrIdx !== undefined ? event.target.attributes[attrIdx].name : null;

    const setting = personnelSettingsProxy.find(s => s.key === key);
    if (!setting) return;

    setting.numberOfSelectedPersonnel += event.target.checked ? 1 : -1;
  }

  async function panelHeadingClickEvent(event) {
    if (
      event.target.classList.contains('personnel-select-button') ||
      event.target.classList.contains('personnel-info-button') ||
      event.target.classList.contains('glyphicon-trash') ||
      event.target.classList.contains('glyphicon-info-sign') ||
      event.target.classList.contains('glyphicon-user')
    ) return;

    let match = event.target.outerHTML.match(/building_id="(\d+)"/);
    if (!match) {
      match = event.target.parentElement.parentElement
        .parentElement.parentElement.outerHTML.match(/building_id="(\d+)"/);
    }
    if (match) await panelHeadingClick(match[1], true);
  }

  async function panelHeadingClick(buildingId, toggle = false) {
    const panelHeading = getPanelHeading(buildingId);
    const panelBody    = getPanelBody(buildingId);
    const href         = panelHeading.outerHTML.match(/href="([^"]+)"/)[1];

    if (loadedBuildings.indexOf(href) > -1) {
      if (toggle) togglePanelBody(panelBody);
      return;
    }

    loadedBuildings.push(href);
    const data = await fetchHrefCached(href);
    panelBody.innerHTML = data;

    Array.from(panelBody.getElementsByClassName('schooling_select_available'))
      .forEach(el => el.parentElement.remove());

    addClickEventHandlerToCheckboxes();
    if (toggle) showPanelBody(panelBody);
  }

  function togglePanelBody(panelBody) {
    panelBody.classList.toggle('hidden');
  }

  function showPanelBody(panelBody) {
    panelBody.classList.remove('hidden');
  }

  function getPanelHeading(buildingId) {
    return document.querySelector(`.personal-select-heading[building_id='${buildingId}']`);
  }

  function getPanelBody(buildingId) {
    return document.querySelector(`.panel-body[building_id='${buildingId}']`);
  }

  // ─── Einstieg ─────────────────────────────────────────────────────────────

  async function main() {
    if (!window.location.href.match(/\/buildings\/\d+\/hire/)) return;

    const h1 = document.querySelector('h1[building_type]');
    const buildingType = h1 ? h1.getAttribute('building_type') : null;
    const isBePo = buildingType === '11';

    // Building-ID aus URL
    const buildingId = window.location.href.match(/\/buildings\/(\d+)\//)[1];

    initPanelBodies();
    removePanelHeadingClickEvent();

    // Generisch, für ALLE Wachenarten: Info-Buttons, Übersicht, Dropdown-Filter
    addPersonnelSelectors(isBePo);
    addQualificationToolbar();

    if (!isBePo) {
      // Auf Nicht-BePo-Wachen ist hier Schluss – die Auto-Anwerbung ist BePo-spezifisch
      return;
    }

    // Ab hier nur für BePo-Wachen: bisherige Auto-Anwerbung mit Ist-Stand-Footer
    addFooter();

    const selectButtons = document.querySelectorAll('.personnel-select-button');
    selectButtons.forEach(b => {
      b.disabled = true;
      b.title = 'Ist-Stand wird geladen …';
    });

    await initEffectiveNeeds(buildingId);

    selectButtons.forEach(b => {
      b.disabled = false;
      b.title = '';
    });
  }

  main();
})();
